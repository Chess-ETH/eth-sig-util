Settlement layer

The desk uses settlement/verify.js before signing, then settlement/signer.js to produce the proposal signature.

const { verify } = require('./settlement/verify');
const { sign } = require('./settlement/signer');

const proposal = verify(request);
const signed = sign(proposal, signer);
